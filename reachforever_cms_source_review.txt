function sameSecret(a, b) {
  if (typeof a !== "string" || typeof b !== "string") return false;
  const left = crypto.createHash("sha256").update(a).digest();
  const right = crypto.createHash("sha256").update(b).digest();
  return crypto.timingSafeEqual(left, right);
}

const SOCIAL_HOSTS = {
  instagram: ["instagram.com"],
  facebook: ["facebook.com", "fb.watch"],
  youtube: ["youtube.com", "youtu.be"],
  tiktok: ["tiktok.com"],
  linkedin: ["linkedin.com"],
  x: ["x.com", "twitter.com"],
  threads: ["threads.net"],
  pinterest: ["pinterest.com"],
};

function validSocialUrl(value, platform) {
  try {
    const url = new URL(value);
    if (url.protocol !== "https:" && url.protocol !== "http:") return false;

    const host = url.hostname.toLowerCase();
    const allowedHosts = platform
      ? SOCIAL_HOSTS[platform] || []
      : Object.values(SOCIAL_HOSTS).flat();

    return allowedHosts.some(
      (domain) => host === domain || host.endsWith(`.${domain}`)
    );
  } catch {
    return false;
  }
}

function parseSocialLinks(value) {
  let links = value;

  if (typeof links === "string") {
    try {
      links = JSON.parse(links);
    } catch {
      return null;
    }
  }

  if (!Array.isArray(links)) return null;

  const cleaned = [];
  for (const link of links) {
    if (!link || typeof link !== "object") return null;

    const platform = String(link.platform || "");
    const url = String(link.url || "").trim();

    if (!SOCIAL_HOSTS[platform] || !validSocialUrl(url, platform)) {
      return null;
    }

    cleaned.push({
      platform,
      url,
      isVisible: link.isVisible !== false,
    });
  }

  return cleaned;
}

function publicResult(result) {
  const legacyLinks = result.platform && result.socialUrl
    ? [{ platform: result.platform, url: result.socialUrl, isVisible: true }]
    : [];

  return {
    _id: result._id,
    imageUrl: result.imageUrl,
    platform: result.platform,
    socialUrl: result.socialUrl,
    socialLinks: Array.isArray(result.socialLinks)
      ? result.socialLinks
      : legacyLinks,
    order: result.order,
  };
}

app.post("/api/admin/login", loginLimiter, async (req, res) => {
  const { username, password } = req.body || {};

  if (
    !process.env.ADMIN_USERNAME ||
    !process.env.ADMIN_PASSWORD ||
    !process.env.JWT_SECRET
  ) {
    return res.status(503).json({ message: "Admin environment variables are not configured." });
  }

  if (
    !sameSecret(username, process.env.ADMIN_USERNAME) ||
    !sameSecret(password, process.env.ADMIN_PASSWORD)
  ) {
    return res.status(401).json({ message: "Incorrect username or password." });
  }

  const token = jwt.sign(
    { role: "admin" },
    process.env.JWT_SECRET,
    { expiresIn: "8h", issuer: "reachforever-api", audience: "reachforever-admin" }
  );

  res.json({ token, expiresIn: 28800 });
});

app.post(
  "/api/admin/results",
  adminAuth,
  upload.single("image"),
  async (req, res, next) => {
    let uploaded;

    try {
      await connectDB();

      if (!req.file) {
        return res.status(400).json({ message: "Choose a cover image." });
      }

      const socialLinks = parseSocialLinks(req.body.socialLinks ?? "[]");
      if (socialLinks === null) {
        return res.status(400).json({
          message: "Check each social platform and enter a valid matching account URL.",
        });
      }

      const firstLink = socialLinks[0];
      uploaded = await uploadImage(req.file.buffer);

      const result = await Result.create({
        imageUrl: uploaded.secure_url,
        imagePublicId: uploaded.public_id,
        platform: firstLink?.platform === "facebook" ? "facebook" : "instagram",
        socialUrl: firstLink?.url || "https://www.instagram.com/",
        socialLinks,
        order: Number.isFinite(Number(req.body.order)) ? Number(req.body.order) : 0,
        isActive: req.body.isActive !== "false",
      });

      res.status(201).json(result);
    } catch (error) {
      if (uploaded?.public_id) {
        try {
          await deleteImage(uploaded.public_id);
        } catch (cleanupError) {
          console.error("Cloudinary rollback cleanup failed:", cleanupError.message);
        }
      }
      next(error);
    }
  }
);

app.patch("/api/admin/results/:id", adminAuth, async (req, res, next) => {
  try {
    await connectDB();

    const updates = {};
    const { platform, socialUrl, socialLinks: submittedLinks, order, isActive } = req.body || {};

    if (platform !== undefined) {
      if (!["instagram", "facebook"].includes(platform)) {
        return res.status(400).json({ message: "Invalid legacy platform." });
      }
      updates.platform = platform;
    }

    if (socialUrl !== undefined) {
      if (!validSocialUrl(socialUrl)) {
        return res.status(400).json({ message: "Enter a valid social account URL." });
      }
      updates.socialUrl = socialUrl;
    }

    if (submittedLinks !== undefined) {
      const socialLinks = parseSocialLinks(submittedLinks);
      if (socialLinks === null) {
        return res.status(400).json({
          message: "Check each social platform and enter a valid matching account URL.",
        });
      }

      updates.socialLinks = socialLinks;
      if (socialLinks.length > 0) {
        updates.platform = socialLinks[0].platform === "facebook"
          ? "facebook"
          : "instagram";
        updates.socialUrl = socialLinks[0].url;
      }
    }

    if (order !== undefined) {
      if (!Number.isFinite(Number(order))) {
        return res.status(400).json({ message: "Order must be a number." });
      }
      updates.order = Number(order);
    }

    if (isActive !== undefined) {
      if (typeof isActive !== "boolean") {
        return res.status(400).json({ message: "isActive must be true or false." });
      }
      updates.isActive = isActive;
    }

    const result = await Result.findByIdAndUpdate(req.params.id, updates, {
      new: true,
      runValidators: true,
    });

    if (!result) return res.status(404).json({ message: "Result not found." });
    res.json(result);
  } catch (error) {
    next(error);
  }
});

app.delete("/api/admin/results/:id", adminAuth, async (req, res, next) => {
  try {
    await connectDB();
    const result = await Result.findById(req.params.id);

    if (!result) return res.status(404).json({ message: "Result not found." });

    await result.deleteOne();

    let imageCleanup = "complete";
    try {
      await deleteImage(result.imagePublicId);
    } catch (error) {
      imageCleanup = "failed";
      console.error("Result deleted from MongoDB, but image cleanup failed:", error.message);
    }

    res.json({
      message: "Result permanently deleted from the database.",
      imageCleanup,
    });
  } catch (error) {
    next(error);
  }
});

app.patch("/api/admin/about-founder", adminAuth, async (req, res, next) => {
  try {
    await connectDB();
    const body = req.body || {};
    const updates = {};

    for (const field of founderFields) {
      if (body[field] === undefined) continue;

      if (field === "isVisible") {
        if (typeof body[field] !== "boolean") {
          return res.status(400).json({ message: "Visibility must be true or false." });
        }
        updates[field] = body[field];
        continue;
      }

      if (typeof body[field] !== "string") {
        return res.status(400).json({ message: `${field} must be text.` });
      }

      const value = body[field].trim();
      if (value.length > 3000) {
        return res.status(400).json({ message: `${field} is too long.` });
      }
      if (field === "buttonUrl" && value &&
          !value.startsWith("#") &&
          !/^https?:\/\//i.test(value)) {
        return res.status(400).json({ message: "Button link must be a section anchor or HTTP(S) URL." });
      }
      updates[field] = value;
    }

    let founder = await AboutFounder.findOne();
    if (!founder) founder = new AboutFounder();
    Object.assign(founder, updates);
    await founder.save();
    res.json(founder);
  } catch (error) {
    next(error);
  }
});

app.use((error, req, res, next) => {
  if (res.headersSent) return next(error);

  if (error instanceof multer.MulterError) {
    const message =
      error.code === "LIMIT_FILE_SIZE"
        ? "Image must be 5 MB or smaller."
        : error.message;
    return res.status(400).json({ message });
  }

  if (error.message?.includes("image") || error.message?.includes("Cloudinary")) {
    return res.status(400).json({ message: error.message });
  }

  console.error(error);
  res.status(500).json({ message: "Server error. Check the backend terminal logs." });
});

const jwt = require("jsonwebtoken");

function adminAuth(req, res, next) {
  const secret = process.env.JWT_SECRET;
  if (!secret) {
    return res.status(503).json({ message: "Admin authentication is not configured." });
  }

  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : "";

  if (!token) {
    return res.status(401).json({ message: "Please sign in as admin." });
  }

  try {
    const payload = jwt.verify(token, secret);
    if (payload.role !== "admin") {
      return res.status(403).json({ message: "Admin access required." });
    }

    req.admin = true;
    next();
  } catch {
    return res.status(401).json({ message: "Session expired. Please sign in again." });
  }
}

module.exports = adminAuth;

const cloudinary = require("cloudinary").v2;

cloudinary.config({
  cloud_name: process.env.CLOUDINARY_CLOUD_NAME,
  api_key: process.env.CLOUDINARY_API_KEY,
  api_secret: process.env.CLOUDINARY_API_SECRET,
});

function isConfigured() {
  return Boolean(
    process.env.CLOUDINARY_CLOUD_NAME &&
    process.env.CLOUDINARY_API_KEY &&
    process.env.CLOUDINARY_API_SECRET
  );
}

function uploadImage(buffer, folder = "reachforever/results") {
  if (!isConfigured()) {
    throw new Error("Cloudinary environment variables are not configured.");
  }

  return new Promise((resolve, reject) => {
    const stream = cloudinary.uploader.upload_stream(
      {
        folder,
        resource_type: "image",
        allowed_formats: ["jpg", "jpeg", "png", "webp", "avif"],
      },
      (error, result) => {
        if (error) return reject(error);
        resolve(result);
      }
    );

    stream.end(buffer);
  });
}

async function deleteImage(publicId) {
  if (!publicId) return;
  if (!isConfigured()) {
    throw new Error("Cannot delete the Cloudinary image: configuration is missing.");
  }

  const result = await cloudinary.uploader.destroy(publicId, {
    resource_type: "image",
    invalidate: true,
  });

  if (result.result !== "ok" && result.result !== "not found") {
    throw new Error(`Cloudinary deletion returned: ${result.result}`);
  }
}

module.exports = { cloudinary, isConfigured, uploadImage, deleteImage };

  ["instagram", "Instagram"],
  ["facebook", "Facebook"],
  ["youtube", "YouTube"],
  ["tiktok", "TikTok"],
  ["linkedin", "LinkedIn"],
  ["x", "X"],

  ["pinterest", "Pinterest"],

function legacyLinks(result) {
  if (Array.isArray(result.socialLinks)) return result.socialLinks;
  return result.platform && result.socialUrl
    ? [{ platform: result.platform, url: result.socialUrl, isVisible: true }]
    : [];
}

export default function AdminPage() {
  const router = useRouter();
  const [token, setToken] = useState("");
  const [ready, setReady] = useState(false);
  const [results, setResults] = useState([]);
  const [socialLinks, setSocialLinks] = useState([]);
  const [order, setOrder] = useState("0");
  const [image, setImage] = useState(null);
  const [editing, setEditing] = useState(null);
  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState("");
  const [error, setError] = useState("");
  const [founderForm, setFounderForm] = useState({
    isVisible: true,
    eyebrow: "THE PEOPLE BEHIND THE WORK",
    heading: "Good work starts with good people.",
    description: "",
    founderName: "Ansh Khatri",
    founderTitle: "Founder & CEO, Reach Forever",
    founderStory: "",
    imageUrl: "",
    buttonText: "Let's grow together",
    buttonUrl: "#contact",
  });
  const [founderImage, setFounderImage] = useState(null);
  const [founderSaving, setFounderSaving] = useState(false);


  useEffect(() => {
    const timer = window.setTimeout(() => {
      setToken(window.localStorage.getItem("reachforeverAdminToken") || "");
      setReady(true);
    }, 0);

    return () => window.clearTimeout(timer);
  }, []);

  const logout = useCallback(() => {
    window.localStorage.removeItem("reachforeverAdminToken");
    setToken("");
    setResults([]);
    router.replace("/admin");
  }, [router]);

  const request = useCallback(async (path, options = {}) => {
    const response = await fetch(`${API_URL}${path}`, {
      ...options,
      headers: {
        ...(options.body instanceof FormData ? {} : { "Content-Type": "application/json" }),
        Authorization: `Bearer ${window.localStorage.getItem("reachforeverAdminToken") || ""}`,
        ...options.headers,
      },
    });

    const data = await response.json().catch(() => ({}));
    if (response.status === 401) {
      logout();
      throw new Error(data.message || "Please sign in again.");
    }
    if (!response.ok) throw new Error(data.message || "Request failed.");
    return data;
  }, [logout]);


  const loadFounder = useCallback(async () => {
    try {
      const data = await request("/api/admin/about-founder");
      setFounderForm((current) => ({ ...current, ...data }));
    } catch (err) {
      setError(err.message);
    }
  }, [request]);

  const loadResults = useCallback(async () => {
    try {
      setResults(await request("/api/admin/results"));
      setError("");
    } catch (err) {
      setError(err.message);
    }
  }, [request]);

  useEffect(() => {
    if (!ready || !token) return;

    const timer = window.setTimeout(() => {
      void loadResults();
      void loadFounder();
    }, 0);

    return () => window.clearTimeout(timer);
  }, [ready, token, loadResults, loadFounder]);


  function updateFounderField(field, value) {
    setFounderForm((current) => ({ ...current, [field]: value }));
  }

  async function saveFounder(event) {
    event.preventDefault();
    setFounderSaving(true);
    setError("");
    setMessage("");

    try {
      const {
        imageUrl,
        imagePublicId,
        ...fields
      } = founderForm;

      await request("/api/admin/about-founder", {
        method: "PATCH",
        body: JSON.stringify(fields),
      });

      if (founderImage) {
        const formData = new FormData();
        formData.append("image", founderImage);

        await request("/api/admin/about-founder/image", {
          method: "POST",
          body: formData,
        });

        setFounderImage(null);
        const input = document.getElementById("founder-portrait");
        if (input) input.value = "";
      }

      await loadFounder();
      setMessage("Founder section saved successfully.");
    } catch (err) {
      setError(err.message);
    } finally {
      setFounderSaving(false);
    }
  }
  function resetForm() {
    setEditing(null);
    setSocialLinks([]);
    setOrder(String(results.length));
    setImage(null);
    const fileInput = document.getElementById("result-cover");
    if (fileInput) fileInput.value = "";
  }

  function beginEdit(result) {
    setEditing(result);
    setSocialLinks(legacyLinks(result).map((link) => ({
      platform: link.platform,
      url: link.url || "",
      isVisible: link.isVisible !== false,

  }

  function updateSocialLink(index, changes) {
    setSocialLinks((current) => current.map((link, i) =>
      i === index ? { ...link, ...changes } : link
    ));
  }

  async function saveResult(event) {
    event.preventDefault();
    setBusy(true);
    setError("");
    setMessage("");

    try {
      const cleanLinks = socialLinks
        .filter((link) => link.url.trim())
        .map((link) => ({
          platform: link.platform,
          url: link.url.trim(),
          isVisible: link.isVisible !== false,
        }));

      for (const link of cleanLinks) {
        let parsed;
        try {
          parsed = new URL(link.url);
        } catch {
          throw new Error(`Enter a valid URL for ${link.platform}.`);
        }
        if (!["http:", "https:"].includes(parsed.protocol)) {
          throw new Error(`Use an HTTP or HTTPS URL for ${link.platform}.`);
        }
      }

      if (editing) {
        await request(`/api/admin/results/${editing._id}`, {
          method: "PATCH",
          body: JSON.stringify({ socialLinks: cleanLinks, order: Number(order) }),
        });

        if (image) {
          const form = new FormData();
          form.append("image", image);
          await request(`/api/admin/results/${editing._id}/image`, {
            method: "POST",
            body: form,
          });
        }
        setMessage("Result updated.");

        if (!image) throw new Error("Choose a cover image first.");
        const form = new FormData();
        form.append("image", image);
        form.append("socialLinks", JSON.stringify(cleanLinks));
        form.append("order", order);
        await request("/api/admin/results", { method: "POST", body: form });
        setMessage("Result added.");
      }

      resetForm();
      await loadResults();
    } catch (err) {
      setError(err.message);
    } finally {
      setBusy(false);
    }
  }

  async function toggleActive(result) {
    setError("");
    setMessage("");
    try {
      await request(`/api/admin/results/${result._id}`, {
        method: "PATCH",
        body: JSON.stringify({ isActive: !result.isActive }),
      });
      setMessage(result.isActive ? "Result hidden from the homepage." : "Result published.");
      await loadResults();
    } catch (err) {
      setError(err.message);
    }
  }

  async function moveResult(result, offset) {
    const sorted = [...results].sort((a, b) => a.order - b.order);
    const index = sorted.findIndex((item) => item._id === result._id);
    const target = index + offset;
    if (target < 0 || target >= sorted.length) return;

    try {
      const first = sorted[index];
      const second = sorted[target];
      await Promise.all([
        request(`/api/admin/results/${first._id}`, {
          method: "PATCH",
          body: JSON.stringify({ order: second.order }),
        }),
        request(`/api/admin/results/${second._id}`, {
          method: "PATCH",
          body: JSON.stringify({ order: first.order }),
        }),
      ]);
      await loadResults();
    } catch (err) {
      setError(err.message);
    }
  }

    setMessage("");

        ? "Result deleted, but Cloudinary cleanup needs attention. Check backend logs."
        : "Result permanently deleted.");
      if (editing?._id === result._id) resetForm();
      await loadResults();
    } catch (err) {
      setError(err.message);
    }
  }

  useEffect(() => {
    if (ready && !token) router.replace("/admin");
  }, [ready, token, router]);

  );

function MovingRow({ results, direction, duration }) {
  const trackRef = useRef(null);
  const pointerRef = useRef(null);
  const draggedRef = useRef(false);
  const duplicated = useMemo(() => [...results, ...results], [results]);

  function startDrag(event) {
    if (event.pointerType === "mouse" && event.button !== 0) return;
    pointerRef.current = { x: event.clientX, lastX: event.clientX };
    draggedRef.current = false;
    trackRef.current?.setPointerCapture?.(event.pointerId);
  }

  function moveDrag(event) {
    if (!pointerRef.current || !trackRef.current) return;
    const delta = event.clientX - pointerRef.current.x;
    if (Math.abs(delta) > 5 && !draggedRef.current) {
      draggedRef.current = true;
      trackRef.current.classList.add("is-dragging");
      trackRef.current.style.setProperty("--drag-shift", "0px");
    }
    pointerRef.current.lastX = event.clientX;
    if (draggedRef.current) {
      trackRef.current.style.setProperty("--drag-shift", `${delta}px`);
    }
  }

  function endDrag() {
    if (trackRef.current && draggedRef.current) {
      trackRef.current.classList.remove("is-dragging");
      trackRef.current.style.removeProperty("--drag-shift");
    }
    pointerRef.current = null;
    draggedRef.current = false;
  }

  return (
    <div className="clean-results-lane">
      <div
        ref={trackRef}
        className={`clean-results-belt ${direction === "move-left" ? "clean-results-belt--left" : "clean-results-belt--right"}`}
        style={{ "--results-duration": duration }}
        onPointerDown={startDrag}
        onPointerMove={moveDrag}
        onPointerUp={endDrag}
        onPointerCancel={endDrag}
        onLostPointerCapture={endDrag}
      >
        {duplicated.map((result, index) => (
          <div className="clean-results-tile" key={`${result._id}-${index}`}>
            <ResultCard result={result} />
          </div>
        ))}
      </div>
    </div>
  );
}

export default function ResultsSection() {
  const [results, setResults] = useState([]);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    let cancelled = false;

    fetch(`${API_URL}/api/results`, { cache: "no-store" })
      .then((response) => {
        if (!response.ok) throw new Error("Could not load results.");
        return response.json();
      })
      .then((data) => {
        if (!cancelled) setResults(Array.isArray(data) ? data : []);
      })
      .catch(() => {
        if (!cancelled) setResults([]);
      })
      .finally(() => {
        if (!cancelled) setLoaded(true);
      });

    return () => {
      cancelled = true;
    };
  }, []);

  return (
    <section className="results-section" id="results">
      <div className="section-shell">
        <div className="results-heading">
          <span className="eyebrow">
            <span className="eyebrow-dot" />
            REAL WORK. REAL RESULTS.
          </span>
          <h2>Built on trust. <span>Focused on growth.</span></h2>
          <p>A closer look at the work and results we share.</p>
        </div>

        {results.length > 0 ? (
          <div className="clean-results-carousel" aria-label="ReachForever results">
            <MovingRow results={results} direction="move-right" duration="180s" />
            <MovingRow results={[...results].reverse()} direction="move-left" duration="195s" />
          </div>
        ) : (
          <div className="results-empty">
            {loaded ? "Our latest results will appear here soon." : "Loading results…"}
          </div>
        )}
      </div>
    </section>
  );
}

import { useEffect, useState } from "react";

const API_URL = (process.env.NEXT_PUBLIC_API_URL || "http://localhost:5000").replace(/\/$/, "");

  isVisible: true,
  eyebrow: "THE PEOPLE BEHIND THE WORK",
  heading: "Good work starts with good people.",
  description:
    "Reach Forever is a full-service digital marketing agency dedicated to helping businesses build meaningful digital experiences that drive measurable growth.",
  founderName: "Ansh Khatri",
  founderTitle: "Founder & CEO, Reach Forever",
  founderStory:

  imageUrl: "",
  buttonText: "Let's grow together",
  buttonUrl: "#contact",
};

export default function AboutFounderSection() {
  const [content, setContent] = useState(DEFAULT_CONTENT);

  useEffect(() => {
    const controller = new AbortController();

    fetch(`${API_URL}/api/about-founder`, {
      cache: "no-store",
      signal: controller.signal,
    })
      .then((response) => {
        if (!response.ok) throw new Error("Founder content unavailable");
        return response.json();
      })
      .then((data) => {
        if (data) setContent({ ...DEFAULT_CONTENT, ...data });
      })
      .catch((error) => {
        if (error.name !== "AbortError") {
          console.error("Unable to load founder section:", error.message);
        }
      });

    return () => controller.abort();
  }, []);

  if (!content.isVisible) return null;

  return (
    <section className="founder-section" id="about-founder">
      <div className="section-shell founder-layout">
        <div className="founder-copy">
          <span className="eyebrow">
